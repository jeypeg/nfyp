	\section{Introduction}
	
	\begin{frame}
		\frametitle{Background and Problem Statement}
		Feed-based recommendation engines drive modern platforms:
		\begin{itemize}
			\item Explicitly engineered around retention and engagement optimization
			\item Continuous architectures lead to digital burnout and reduced user agency
		\end{itemize}
		
		Limitations of existing interventions:
		\begin{itemize}
			\item Rely almost entirely on external interface controls (e.g., app timers)
			\item Recommendation algorithms themselves remain unexamined as tools for self-regulation
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{The "Not For You Page" (NFYP)}
		Core concept and objectives:
		\begin{itemize}
			\item An alternative feed model that inverts retention objectives
			\item Systematically reduces content resonance and introduces intentional friction
			\item Evaluates if recommendation logic can directly support voluntary platform exit
		\end{itemize}
		
		Research Goal:
		\begin{itemize}
			\item Bridge the gap between digital well-being interventions and underlying algorithmic logic
		\end{itemize}
	\end{frame}
	
	\section{Related Work}
	
	\begin{frame}
		\frametitle{Related Work}
		Engagement-First Algorithmic Design:
		\begin{itemize}
			\item TikTok-style feeds rely on collaborative and content-based filtering to maximize retention (Zhou, 2024)
			\item Documented negative outcomes include compulsive use, echo chambers, and identity formation (Smith \& Short, 2022; Kuncoro \& Hasanah, 2024)
		\end{itemize}
		
		Limitations of Existing Well-Being Interventions:
		\begin{itemize}
			\item Current tools focus almost exclusively on interface-level friction or external timers (Haliburton et al., 2024; Almoallim \& Sas, 2022)
			\item \textbf{The Gap:} Existing interventions leave the underlying recommendation objective entirely unexamined
		\end{itemize}
	\end{frame}
	
	\section{Methodology}
	
	\begin{frame}
		\frametitle{System Mechanics and Experimental Setup}
		System Mechanics:
		\begin{itemize}
			\item Initial profiling assessment establishes baseline interest vectors
			\item Algorithm systematically suppresses primary interests and samples uneventful media
		\end{itemize}
		
		Experimental Design:
		\begin{itemize}
			\item Sample of 40 active daily users ($N = 40$), naive to disengagement hypothesis
			\item Standardized mobile test environment logging detailed interaction telemetry
			\item 1:1 randomized assignment to Control (standard feed) vs. Treatment (NFYP)
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Disengagement Dynamics and Exit Motivations}
		Retention and Exit Analysis:
		\begin{itemize}
			\item Kaplan-Meier survival curves show clear retention divergence
			\item At 30 minutes, 34\% of NFYP participants exited vs. 16\% in control
		\end{itemize}
		
		Self-Reported Exit Motivations:
		\begin{itemize}
			\item NFYP Group: 54\% cited content irrelevance or mild disinterest
			\item Control Group: 64\% cited external schedule constraints
		\end{itemize}
	\end{frame}
	
	\section{Discussion}
	
	\begin{frame}
		\frametitle{Discussion and Implications}
		Key Takeaways:
		\begin{itemize}
			\item Algorithmic friction successfully prompts voluntary session termination
			\item Demonstrates feasibility of embedding well-being objectives into core recommendation logic
		\end{itemize}
		
		Limitations and Future Work:
		\begin{itemize}
			\item Evaluation limited to single-session setup with $N = 40$ sample size
			\item Long-term user adaptation and habituation remain to be explored
			\item Need for research on longitudinal adaptation and dynamic friction mechanisms
		\end{itemize}
	\end{frame}
